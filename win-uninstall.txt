# Khatt al-Quran: remove its fonts and the Word add-in (Windows).
#   irm https://quran.abdulloh.org/win-uninstall.txt | iex
#   $env:KHATT_ONLY='word-tab'; irm https://quran.abdulloh.org/win-uninstall.txt | iex   (only the ribbon tab)
# The "Khatt al-Quran" ribbon tab: KhattAlQuran.dotm of Word's STARTUP folder goes into the Recycle Bin (a running Word
# keeps it open: then close Word and run this again).
$TabDirs = @(Join-Path $env:APPDATA 'Microsoft\Word\STARTUP')
try {
  $k = [Microsoft.Win32.Registry]::CurrentUser.OpenSubKey('Software\Microsoft\Office\16.0\Word\Options')
  if ($k) { $v = [string]$k.GetValue('STARTUP-PATH'); $k.Close(); if ($v) { $TabDirs += [Environment]::ExpandEnvironmentVariables($v.TrimEnd('\')) } }
} catch {}
foreach ($d in ($TabDirs | Select-Object -Unique)) {
  $p = Join-Path $d 'KhattAlQuran.dotm'
  if (Test-Path -LiteralPath $p) {
    try {
      Add-Type -AssemblyName Microsoft.VisualBasic
      [Microsoft.VisualBasic.FileIO.FileSystem]::DeleteFile($p, 'OnlyErrorDialogs', 'SendToRecycleBin')
      Write-Host "Khatt al-Quran ribbon tab: $p is in the Recycle Bin. Restart Word." -ForegroundColor Green
    } catch { Write-Host "Close Word, then run this again to remove $p" -ForegroundColor Yellow }
  }
}
if (([string]$env:KHATT_ONLY).Trim() -eq 'word-tab') { return }
$Clock = [Diagnostics.Stopwatch]::StartNew()
$Key = [Microsoft.Win32.Registry]::CurrentUser.CreateSubKey('Software\Microsoft\Windows NT\CurrentVersion\Fonts')
Add-Type -Namespace MyQuran -Name GdiU -MemberDefinition @'
[DllImport("gdi32.dll", CharSet = CharSet.Unicode)] public static extern int RemoveFontResourceW(string file);
[DllImport("user32.dll")] public static extern int SendMessageTimeoutW(System.IntPtr h, uint m, System.IntPtr w, System.IntPtr l, uint f, uint t, out System.IntPtr r);
'@
# switch the fonts off first (8 at a time): loaded fonts are locked and their folder could not be deleted
$files = @()
foreach ($n in @($Key.GetValueNames() | Where-Object { $_ -like 'MyQuran *' })) { $files += [string]$Key.GetValue($n); $Key.DeleteValue($n, $false) }
$Key.Close()
if ($files.Count) {
  $pool = [RunspaceFactory]::CreateRunspacePool(1, 8); $pool.Open(); $jobs = @()
  for ($k = 0; $k -lt 8; $k++) {
    $batch = @(for ($i = $k; $i -lt $files.Count; $i += 8) { $files[$i] })
    if (-not $batch.Count) { continue }
    $ps = [PowerShell]::Create().AddScript({ param($b) foreach ($f in $b) { [void][MyQuran.GdiU]::RemoveFontResourceW($f) } }).AddArgument($batch)
    $ps.RunspacePool = $pool; $jobs += [pscustomobject]@{ P = $ps; H = $ps.BeginInvoke() }
  }
  while (@($jobs | Where-Object { -not $_.H.IsCompleted }).Count) {
    Write-Progress -Activity 'Removing Khatt al-Quran fonts' -Status ('{0} fonts   {1:mm\:ss}' -f $files.Count, $Clock.Elapsed)
    Start-Sleep -Milliseconds 300
  }
  foreach ($j in $jobs) { $j.P.Dispose() }; $pool.Close()
  Write-Progress -Activity 'Removing Khatt al-Quran fonts' -Completed
}
$r = [IntPtr]::Zero
[void][MyQuran.GdiU]::SendMessageTimeoutW([IntPtr]0xffff, 0x001D, [IntPtr]::Zero, [IntPtr]::Zero, 2, 1000, [ref]$r)  # WM_FONTCHANGE
$Dev = [Microsoft.Win32.Registry]::CurrentUser.OpenSubKey('Software\Microsoft\Office\16.0\WEF\Developer', $true)
if ($Dev) { $Dev.DeleteValue('5392de1f-c12d-4292-8fb5-64c79333149e', $false); $Dev.Close() }
# the updater of the add-in manifest (win.txt): its Run value and daily task go, its files into the Recycle Bin
try { $Run = [Microsoft.Win32.Registry]::CurrentUser.OpenSubKey('Software\Microsoft\Windows\CurrentVersion\Run', $true); if ($Run) { $Run.DeleteValue('KhattAlQuranManifest', $false); $Run.Close() } } catch {}
try { Unregister-ScheduledTask -TaskName 'Khatt al-Quran manifest' -Confirm:$false -ErrorAction Stop } catch {}
try {
  Add-Type -AssemblyName Microsoft.VisualBasic
  foreach ($f in 'khatt-manifest.ps1', 'khatt-manifest.vbs', 'manifest.log', 'manifest.xml.bak') {
    $p = Join-Path (Join-Path $env:LOCALAPPDATA 'MyQuran') $f
    if (Test-Path -LiteralPath $p) { [Microsoft.VisualBasic.FileIO.FileSystem]::DeleteFile($p, 'OnlyErrorDialogs', 'SendToRecycleBin') }
  }
} catch {}
Remove-Item -Recurse -Force (Join-Path $env:LOCALAPPDATA 'MyQuran') -ErrorAction SilentlyContinue
Write-Host ('Khatt al-Quran removed ({0} fonts, {1:mm\:ss}). Restart Word.' -f $files.Count, $Clock.Elapsed) -ForegroundColor Green
if (Test-Path (Join-Path $env:LOCALAPPDATA 'MyQuran')) { Write-Host 'Some font files were in use: sign out and in, then run this command again to delete them.' -ForegroundColor Yellow }
