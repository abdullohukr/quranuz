# Khatt al-Quran installer for Windows: all fonts (incl. QPC V1/V2/V4 page fonts) and the Word add-in.
# No administrator rights needed. Run in PowerShell:
#   irm https://quran.abdulloh.org/win.txt | iex
# Fonts go to their own folder %LOCALAPPDATA%\MyQuran\Fonts and are registered for the
# current user. Every run is a clean reinstall (close Word first). Uninstall:  irm https://quran.abdulloh.org/win-uninstall.txt | iex
$ErrorActionPreference = 'Stop'
$ProgressPreference = 'Continue'
[Net.ServicePointManager]::SecurityProtocol = [Net.ServicePointManager]::SecurityProtocol -bor [Net.SecurityProtocolType]::Tls12
[Net.ServicePointManager]::DefaultConnectionLimit = 16   # .NET allows only 2 connections per server by default
$Site = 'https://quran.abdulloh.org'
$Mirror = 'https://abdullohukr.github.io/quranuz'   # the same site on GitHub Pages
$Rel  = 'https://github.com/abdullohukr/quranuz/releases/download/fonts'   # QPC packages; our site is the fallback
$Home2 = Join-Path $env:LOCALAPPDATA 'MyQuran'
$FontDir = Join-Path $Home2 'Fonts'
$FontKeyPath = 'Software\Microsoft\Windows NT\CurrentVersion\Fonts'
$UserFonts = Join-Path $env:LOCALAPPDATA 'Microsoft\Windows\Fonts'
$SysFonts = Join-Path $env:WINDIR 'Fonts'
$Old = Join-Path $Home2 'OldFonts'
# a temporary folder whose path has only plain ASCII letters: a user name in Cyrillic (C:\Users\Админ\...)
# or with spaces made unpacking and removing the downloads fail on some computers
$TmpBase = $env:TEMP
if ($TmpBase -match '[^\x21-\x7E]') { $TmpBase = Join-Path $env:SystemDrive 'ProgramData' }
$Tmp = Join-Path $TmpBase ('KhattTmp-' + [guid]::NewGuid().ToString('N').Substring(0, 8))
New-Item -ItemType Directory -Force -Path $FontDir, $Tmp | Out-Null
$FontKey = [Microsoft.Win32.Registry]::CurrentUser.CreateSubKey($FontKeyPath)   # direct access: far faster than New-ItemProperty
Add-Type -AssemblyName System.IO.Compression.FileSystem
try { $Host.UI.RawUI.WindowTitle = 'Khatt al-Quran' } catch {}

# Output: every step is a coloured heading with its number and the time since the start, then one line per
# language (English, Uzbek, Russian; the Windows console cannot show Arabic letters joined and right to left).
$Clock = [Diagnostics.Stopwatch]::StartNew()
$Steps = 6
function T { '{0:00}:{1:00}' -f [int][Math]::Floor($Clock.Elapsed.TotalMinutes), $Clock.Elapsed.Seconds }
function Step($n, $en, $uz, $ru) {
  Write-Host ''
  Write-Host ("─── $n/$Steps " + ('─' * 38) + ' ' + (T)) -ForegroundColor Cyan
  Write-Host "  $en" -ForegroundColor White
  Write-Host "  $uz" -ForegroundColor Gray
  Write-Host "  $ru" -ForegroundColor Gray
}
function Ok($t) { Write-Host "  √ $t" -ForegroundColor Green }
function Warn($t) { Write-Host "  ! $t" -ForegroundColor Yellow }

# A click inside the console window starts a text selection and PAUSES the script until Enter or Esc
# ("QuickEdit"): it looked as if the installer hung. QuickEdit is off while the installer runs.
Add-Type -Namespace MyQuran -Name Con -MemberDefinition @'
[DllImport("kernel32.dll")] public static extern System.IntPtr GetStdHandle(int h);
[DllImport("kernel32.dll")] public static extern bool GetConsoleMode(System.IntPtr h, out uint m);
[DllImport("kernel32.dll")] public static extern bool SetConsoleMode(System.IntPtr h, uint m);
'@
$ConIn = [MyQuran.Con]::GetStdHandle(-10); $ConMode = [uint32]0
$QuickEdit = [MyQuran.Con]::GetConsoleMode($ConIn, [ref]$ConMode)
if ($QuickEdit) { [void][MyQuran.Con]::SetConsoleMode($ConIn, ($ConMode -band (-bnot [uint32]0x40)) -bor 0x80) }
function Restore-Console { if ($QuickEdit) { [void][MyQuran.Con]::SetConsoleMode($ConIn, $ConMode) } }

if (Get-Process WINWORD -ErrorAction SilentlyContinue) {
  Warn 'Close Word first, then press Enter'
  Write-Host "    Word'ni yoping, so'ng Enter tugmasini bosing"
  Write-Host '    Закройте Word и нажмите Enter'
  Read-Host | Out-Null
}

# make new fonts usable right away (otherwise only after the next sign-in)
Add-Type -Namespace MyQuran -Name Gdi -MemberDefinition @'
[DllImport("gdi32.dll", CharSet = CharSet.Unicode)] public static extern int AddFontResourceW(string file);
[DllImport("gdi32.dll", CharSet = CharSet.Unicode)] public static extern int RemoveFontResourceW(string file);
[DllImport("user32.dll")] public static extern int SendMessageTimeoutW(System.IntPtr h, uint m, System.IntPtr w, System.IntPtr l, uint f, uint t, out System.IntPtr r);
'@

# Runs $Work over $Items in 8 threads; $Work gets (batch of items, shared state) and counts done items in $st.n.
# $Measure (optional) returns @(done, total, unit) for the progress bar, e.g. megabytes instead of pieces.
function Invoke-Parallel($Items, [scriptblock]$Work, $Label, [scriptblock]$Measure) {
  $Items = @($Items)
  if (-not $Items.Count) { return }
  $st = [hashtable]::Synchronized(@{ n = 0 })
  $pool = [RunspaceFactory]::CreateRunspacePool(1, 8); $pool.Open()
  $jobs = @()
  for ($k = 0; $k -lt 8; $k++) {
    $batch = @(for ($i = $k; $i -lt $Items.Count; $i += 8) { $Items[$i] })
    if (-not $batch.Count) { continue }
    $ps = [PowerShell]::Create().AddScript($Work).AddArgument($batch).AddArgument($st)
    $ps.RunspacePool = $pool
    $jobs += [pscustomobject]@{ P = $ps; H = $ps.BeginInvoke() }
  }
  $watch = [Diagnostics.Stopwatch]::StartNew()
  while (@($jobs | Where-Object { -not $_.H.IsCompleted }).Count) {
    $m = if ($Measure) { & $Measure } else { @($st.n, $Items.Count, '') }
    $done = [double]$m[0]; $total = [Math]::Max(1, [double]$m[1])
    $left = if ($done -gt 0) { [int]($watch.Elapsed.TotalSeconds * ($total - $done) / $done) } else { -1 }
    $txt = '{0:N0} / {1:N0}{2}   elapsed {3}' -f $done, $total, $m[2], (T)
    if ($left -ge 0) { Write-Progress -Activity $Label -Status $txt -PercentComplete ([Math]::Min(100, [int](100 * $done / $total))) -SecondsRemaining $left }
    else { Write-Progress -Activity $Label -Status $txt -PercentComplete 0 }
    Start-Sleep -Milliseconds 300
  }
  foreach ($j in $jobs) { try { $j.P.EndInvoke($j.H) | Out-Null } catch {} ; $j.P.Dispose() }
  $pool.Close()
  Write-Progress -Activity $Label -Completed
}

# one download with a progress bar (the fallback when the parallel download did not give a valid zip)
function Get-File($url, $out, $label) {
  $res = [System.Net.HttpWebRequest]::Create($url).GetResponse()
  $total = $res.ContentLength; $in = $res.GetResponseStream(); $fs = [IO.File]::Create($out)
  $buf = New-Object byte[] 262144; $done = 0; $shown = 0
  try {
    while (($n = $in.Read($buf, 0, $buf.Length)) -gt 0) {
      $fs.Write($buf, 0, $n); $done += $n
      if ($total -gt 0 -and ($done - $shown) -ge 512KB) {
        $shown = $done
        Write-Progress -Activity $label -Status ('{0:N1} / {1:N1} MB' -f ($done / 1MB), ($total / 1MB)) -PercentComplete ([int](100 * $done / $total))
      }
    }
  } finally { $fs.Close(); $in.Close(); $res.Close(); Write-Progress -Activity $label -Completed }
}
function Test-Zip($f) { try { [IO.Compression.ZipFile]::OpenRead($f).Dispose(); $true } catch { $false } }
function Get-Zip($out, [string[]]$urls) {
  foreach ($u in $urls) {
    try { Get-File $u $out ("Downloading " + (Split-Path $u -Leaf)); if (Test-Zip $out) { return $true } } catch {}
    Warn "could not get a valid zip from $u"
  }
  Remove-Item -Force $out -ErrorAction SilentlyContinue
  return $false
}

Step 1 'Downloading the fonts (8 connections)' 'Shriftlar yuklab olinmoqda' 'Скачивание шрифтов'
# Everything is downloaded and checked BEFORE the installed fonts are touched. The servers give about 1 MB/s
# per connection: every package is fetched in 16 MB pieces, 8 at a time; a package whose pieces do not make
# a valid zip is downloaded again in one piece (with the fallbacks).
$Pkgs = [ordered]@{
  'main.zip'   = @("$Site/fonts/MyQuran-fonts.zip", "$Mirror/fonts/MyQuran-fonts.zip")
  'qpc-V1.zip' = @("$Rel/MyQuran-QPC-V1-fonts.zip", "$Site/fonts/MyQuran-QPC-V1-fonts.zip")
  'qpc-V2.zip' = @("$Rel/MyQuran-QPC-V2-fonts.zip", "$Site/fonts/MyQuran-QPC-V2-fonts.zip")
  'qpc-V4.zip' = @("$Rel/MyQuran-QPC-V4-fonts.zip", "$Site/fonts/MyQuran-QPC-V4-fonts.zip")
}
# A QPC package that has not changed since the last installation (same size and ETag) and whose 604 fonts
# are all in place is kept and not downloaded again.
$StateFile = Join-Path $Home2 'packages.txt'
$Before = @(); if (Test-Path $StateFile) { $Before = @(Get-Content $StateFile) }
$parts = @(); $sizes = @{}; $Heads = @{}; $Unchanged = @()
foreach ($k in $Pkgs.Keys) {
  $size = 0; $etag = ''
  try { $r = [Net.HttpWebRequest]::Create($Pkgs[$k][0]); $r.Method = 'HEAD'; $resp = $r.GetResponse(); $size = $resp.ContentLength; $etag = [string]$resp.Headers['ETag']; $resp.Close() } catch {}
  $Heads[$k] = "$k $size $etag"
  if ($k -match '^qpc-(V\d)' -and $size -gt 0 -and $Before -contains $Heads[$k] -and
      @(Get-ChildItem $FontDir -Filter "QPC-$($Matches[1])-p*.ttf" -ErrorAction SilentlyContinue).Count -eq 604) {
    $Unchanged += $Matches[1]; continue
  }
  $sizes[$k] = [Math]::Max([long]0, [long]$size)
  for ($i = 0; $size -gt 0 -and $i * 16MB -lt $size; $i++) {
    $parts += [pscustomobject]@{ Url = $Pkgs[$k][0]; Out = (Join-Path $Tmp ('{0}.{1:D3}' -f $k, $i)); From = [long]($i * 16MB); To = [long]([Math]::Min($size, ($i + 1) * 16MB) - 1) }
  }
}
$Total = ($sizes.Values | Measure-Object -Sum).Sum
Invoke-Parallel $parts {
  param($batch, $st)
  foreach ($p in $batch) {
    for ($try = 0; $try -lt 4; $try++) {
      try {
        $req = [Net.HttpWebRequest]::Create($p.Url); $req.AddRange($p.From, $p.To); $req.Timeout = 60000; $req.ReadWriteTimeout = 60000
        $res = $req.GetResponse()
        if ([int]$res.StatusCode -ne 206) { $res.Close(); break }   # the range was ignored: leave it to the one-piece download
        $in = $res.GetResponseStream(); $fs = [IO.File]::Create($p.Out)
        try { $in.CopyTo($fs, 262144) } finally { $fs.Close(); $in.Close(); $res.Close() }
        if ((Get-Item $p.Out).Length -eq ($p.To - $p.From + 1)) { break }
      } catch { Start-Sleep -Seconds 2 }
    }
    $st.n++
  }
} 'Downloading fonts (8 connections)' {
  $d = 0; foreach ($f in [IO.Directory]::GetFiles($Tmp, '*.zip.*')) { $d += ([IO.FileInfo]$f).Length }
  @(($d / 1MB), ($Total / 1MB), ' MB')
}
foreach ($k in $Pkgs.Keys) {
  $out = Join-Path $Tmp $k
  if ($k -match '^qpc-(V\d)' -and $Unchanged -contains $Matches[1]) { continue }
  $ps = @([IO.Directory]::GetFiles($Tmp, "$k.*") | Where-Object { $_ -match '\.\d{3}$' } | Sort-Object)
  if ($sizes[$k] -gt 0 -and $ps.Count) {
    $fs = [IO.File]::Create($out)
    foreach ($p in $ps) { $in = [IO.File]::OpenRead($p); $in.CopyTo($fs); $in.Close(); Remove-Item -Force $p }
    $fs.Close()
    if ((Get-Item $out).Length -ne $sizes[$k] -or -not (Test-Zip $out)) { Remove-Item -Force $out }
  }
  if (-not (Test-Path $out)) { [void](Get-Zip $out $Pkgs[$k]) }
}
$MainZip = Join-Path $Tmp 'main.zip'
if (-not (Test-Path $MainZip)) {
  Write-Host ''
  Write-Host '  The fonts could not be downloaded. Nothing was changed. Check the internet connection and run the command again.' -ForegroundColor Red
  Write-Host "  Shriftlar yuklab olinmadi, hech narsa o'zgartirilmadi. Internetni tekshirib, buyruqni qayta ishga tushiring." -ForegroundColor Red
  Write-Host '  Шрифты не скачались, ничего не изменено. Проверьте интернет и запустите команду ещё раз.' -ForegroundColor Red
  Remove-Item -Recurse -Force $Tmp -ErrorAction SilentlyContinue; Restore-Console; return
}
$got = 0; foreach ($k in $Pkgs.Keys) { $f = Join-Path $Tmp $k; if (Test-Path $f) { $got += (Get-Item $f).Length } }
Ok ('{0:N0} MB downloaded' -f ($got / 1MB))
# QPC packages whose installed fonts stay: unchanged, or not downloaded now (then the previous ones are kept)
$Keep = @()
foreach ($v in 'V1', 'V2', 'V4') {
  if ($Unchanged -contains $v) { $Keep += $v; Ok "QPC $v unchanged, already installed: not downloaded again" }
  elseif (-not (Test-Path (Join-Path $Tmp "qpc-$v.zip"))) { $Keep += $v; Warn "QPC $v not downloaded now; the fonts installed before are kept" }
}
function Test-Kept($file) { $file -match 'QPC-(V\d)-' -and $Keep -contains $Matches[1] }

Step 2 'Removing the previous installation' "Oldingi o'rnatish olib tashlanmoqda" 'Удаление прежней установки'
# Clean reinstall: unregister and delete the previous MyQuran fonts; copies of the same font
# files installed by hand for this user are moved to %LOCALAPPDATA%\MyQuran\OldFonts.
$unload = @()
foreach ($name in $FontKey.GetValueNames()) {
  $path = [string]$FontKey.GetValue($name)
  $ours = $name -like 'MyQuran *'
  $mine = $path -like "$UserFonts\*" -and (Test-Path (Join-Path $FontDir (Split-Path $path -Leaf)))
  # an entry whose file is gone (deleted by hand, older installs) makes Word say
  # "We weren't able to load all of your fonts": remove the entry
  $gone = $path -and [IO.Path]::IsPathRooted($path) -and -not (Test-Path $path)
  if ($ours -and -not $gone -and (Test-Kept $name)) { continue }   # a kept package stays registered and switched on
  if ($ours -or $mine -or $gone) {
    $unload += $path
    $FontKey.DeleteValue($name, $false)
    if ($mine -and (Test-Path $path)) {
      New-Item -ItemType Directory -Force -Path $Old | Out-Null
      Move-Item -Force $path $Old -ErrorAction SilentlyContinue
    }
  }
}
Invoke-Parallel $unload { param($batch, $st) foreach ($f in $batch) { [void][MyQuran.Gdi]::RemoveFontResourceW($f); $st.n++ } } 'Removing the previous fonts'
Get-ChildItem $FontDir -File -ErrorAction SilentlyContinue | Where-Object { -not (Test-Kept $_.Name) } | Remove-Item -Force -ErrorAction SilentlyContinue
Ok "$($unload.Count) previous fonts removed   ($(T))"

# Fonts are unpacked straight into the font folder and registered for this user; they are switched on
# (AddFontResource) together afterwards, in 8 threads. One by one this took about half a second per font.
$Load = New-Object System.Collections.Generic.List[string]
function Install-Zip($zip) {
  if (-not (Test-Path $zip)) { return }
  $z = [IO.Compression.ZipFile]::OpenRead($zip); $copied = 0; $kept = 0
  try {
    $entries = @($z.Entries | Where-Object { $_.Name -match '\.(ttf|otf)$' }); $i = 0
    foreach ($e in $entries) {
      $i++
      if ($i % 25 -eq 0) { Write-Progress -Activity 'Unpacking fonts' -Status "$i / $($entries.Count)   $(T)" -PercentComplete ([int](100 * $i / $entries.Count)) }
      # already installed for all users (C:\Windows\Fonts): a second copy makes Word complain
      if (Test-Path (Join-Path $SysFonts $e.Name)) { $kept++; continue }
      $dst = Join-Path $FontDir $e.Name
      try { [IO.Compression.ZipFileExtensions]::ExtractToFile($e, $dst, $true) }
      catch { if (-not (Test-Path $dst)) { Warn "$($e.Name): $($_.Exception.Message)"; continue } }   # in use: the same file stays
      $kind = if ($e.Name -match '\.otf$') { ' (OpenType)' } else { ' (TrueType)' }
      $FontKey.SetValue('MyQuran ' + [IO.Path]::GetFileNameWithoutExtension($e.Name) + $kind, $dst)
      $Load.Add($dst); $copied++
    }
  } finally { $z.Dispose(); Write-Progress -Activity 'Unpacking fonts' -Completed }
  Ok ("$copied installed" + $(if ($kept) { ", $kept already installed for all users (kept)" } else { '' }))
}
